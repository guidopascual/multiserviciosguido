Cómo instalarlo en TradingView

1. Click en el botón verde de arriba — el script queda copiado
2. Abre TradingView → gráfico de SPY o QQQ en M15
3. Abajo: Pine Editor → borra todo → pega (Ctrl+V)
4. Click Guardar → Agregar al gráfico
5. Click derecho en el indicador → Crear alerta → elige cada condición (E1 CALL, E1 PUT, etc.)
6. En cada alerta activa: notificación + email · Frecuencia: Una vez por cierre de barra

//@version=5
indicator("📊 Pine Script Para Trading View", overlay=true, max_bars_back=500)

// ══════════════════════════════════════════
//   6 ESTRATEGIAS — VERSIÓN OBJETIVA
//   Ventanas = REQUISITO (no cuentan en score)
//   Strike recomendado: ATM siempre
//   E1: Bollinger 0DTE      vela 9:45 ET      · 3/4 · TP20 SL15
//   E2: Resumen 20 + H1     9:30-10:00 ET     · 3/4 · TP20 SL15
//   E3: Cierre c/dirección  3:30-3:58 ET      · 3/4 · salida 3:55 / SL10
//   E4: Continuación        10:00-11:00 ET    · 3/4 · TP20 SL15
//   E5: Semanal             Lun/Mar           · 4/5 · TP50 SL25
//   E6: Swing (manual, gráfico diario)        · 4/5 · TP15 SL5 trailing
//   Filtro universal: VIX < 25 (verificar manualmente)
//   Regla spread: si (ask-bid) > 5% del contrato, NO entrar
//
//   NOTA: Estrategias sin probar. Validar con
//   paper trading mínimo 2 meses.
// ══════════════════════════════════════════

// ── INPUTS ──
min_e1 = input.int(3, "E1 mínimo (de 4)", group="Mínimos")
min_e2 = input.int(3, "E2 mínimo (de 4)", group="Mínimos")
min_e3 = input.int(3, "E3 mínimo (de 4)", group="Mínimos")
min_e4 = input.int(3, "E4 mínimo (de 4)", group="Mínimos")
min_e5 = input.int(4, "E5 mínimo (de 5)", group="Mínimos")
adx_e4_min = input.int(25, "E4 ADX mínimo", group="ADX")
adx_e5_min = input.int(20, "E5 ADX mínimo", group="ADX")
res_pct = input.float(0.3, "Umbral resistencia/soporte %", group="E2") / 100
mom_pct = input.float(0.3, "E4 momentum mínimo %", group="E4") / 100
show_table = input.bool(true, "Mostrar tabla", group="Visual")
show_sig = input.bool(true, "Mostrar señales", group="Visual")

// ── HORA NY ──
ny_h = hour(time, "America/New_York")
ny_m = minute(time, "America/New_York")
ny_t = ny_h*60 + ny_m
ny_dow = dayofweek(time, "America/New_York")

// Ventanas — REQUISITOS, no condiciones
is_vela945 = (ny_h==9 and ny_m>=45)
w_e2 = ny_t >= (9*60+30) and ny_t < (10*60)
w_e4 = ny_t >= (10*60) and ny_t < (11*60)
w_e3 = ny_t >= (15*60+30) and ny_t < (15*60+58)
is_mon = ny_dow==2
is_tue = ny_dow==3
is_thu = ny_dow==5
w_e5 = (is_mon or is_tue) and ny_t >= (9*60+30) and ny_t < (11*60)

// ── INDICADORES BASE ──
[bb_mid, bb_up, bb_lo] = ta.bb(close, 20, 2)
inside_bb = open > bb_lo and open < bb_up
ema7  = ta.ema(close, 7)
ema26 = ta.ema(close, 26)
vwap_v = ta.vwap(hlc3)
above_vwap = close > vwap_v
below_vwap = close < vwap_v
[dip, dim, adx_v] = ta.dmi(14, 14)
vol_avg = ta.sma(volume, 5)
vol_high = volume > vol_avg

// Cuerpo sólido = cuerpo > 50% del rango
rng = high - low
body = math.abs(close - open)
solid = rng > 0 and body > rng * 0.5
bull_solid = close > open and solid
bear_solid = close < open and solid

// Vela 9:30 previa (para E1) — con cuerpo sólido
v930_bull = close[1] > open[1] and (high[1]-low[1] > 0) and math.abs(close[1]-open[1]) > (high[1]-low[1])*0.5
v930_bear = close[1] < open[1] and (high[1]-low[1] > 0) and math.abs(close[1]-open[1]) > (high[1]-low[1])*0.5

// ── TENDENCIA H1 OBJETIVA (para E2) ──
// EMA20 H1 subió 3 velas H1 consecutivas + precio sobre/bajo ella
ema20_h1_0 = request.security(syminfo.tickerid, "60", ta.ema(close,20))
ema20_h1_1 = request.security(syminfo.tickerid, "60", ta.ema(close,20)[1])
ema20_h1_2 = request.security(syminfo.tickerid, "60", ta.ema(close,20)[2])
ema20_h1_3 = request.security(syminfo.tickerid, "60", ta.ema(close,20)[3])
h1_bull = ema20_h1_0 > ema20_h1_1 and ema20_h1_1 > ema20_h1_2 and ema20_h1_2 > ema20_h1_3 and close > ema20_h1_0
h1_bear = ema20_h1_0 < ema20_h1_1 and ema20_h1_1 < ema20_h1_2 and ema20_h1_2 < ema20_h1_3 and close < ema20_h1_0

// ── RESISTENCIA/SOPORTE OBJETIVO (para E2) ──
pd_high = request.security(syminfo.tickerid, "D", high[1])
pd_low  = request.security(syminfo.tickerid, "D", low[1])
no_res = pd_high > close * (1 + res_pct)   // máx día anterior a más de 0.3% arriba
no_sup = pd_low  < close * (1 - res_pct)   // mín día anterior a más de 0.3% abajo

// ── DIRECCIÓN DEL DÍA (para E3) ──
day_open = request.security(syminfo.tickerid, "D", open)
day_chg = (close - day_open) / day_open
day_up = day_chg > 0.01
day_down = day_chg < -0.01

// ── MOMENTUM DESDE 9:30 (para E4) ──
mom_up = day_chg > mom_pct
mom_down = day_chg < -mom_pct

// ── TENDENCIA DIARIA OBJETIVA (para E5) ──
ema20_d0 = request.security(syminfo.tickerid, "D", ta.ema(close,20))
ema20_d1 = request.security(syminfo.tickerid, "D", ta.ema(close,20)[1])
ema20_d2 = request.security(syminfo.tickerid, "D", ta.ema(close,20)[2])
ema20_d3 = request.security(syminfo.tickerid, "D", ta.ema(close,20)[3])
wk_bull = ema20_d0 > ema20_d1 and ema20_d1 > ema20_d2 and ema20_d2 > ema20_d3 and close > ema20_d0
wk_bear = ema20_d0 < ema20_d1 and ema20_d1 < ema20_d2 and ema20_d2 < ema20_d3 and close < ema20_d0

// ── RANGO SEMANA ANTERIOR (requisito E5: < 3%) ──
pw_high = request.security(syminfo.tickerid, "W", high[1])
pw_low  = request.security(syminfo.tickerid, "W", low[1])
pw_range_ok = pw_low > 0 and (pw_high - pw_low) / pw_low < 0.03

// ── StochRSI (para E5): cruce alcista bajo 50 en últimas 3 velas ──
rsi_s = ta.rsi(close,14)
stoch_raw = ta.stoch(rsi_s, rsi_s, rsi_s, 14)
stoch_k = ta.sma(stoch_raw, 3)
stoch_d = ta.sma(stoch_k, 3)
cross_up = ta.crossover(stoch_k, stoch_d) and stoch_k < 50
cross_dn = ta.crossunder(stoch_k, stoch_d) and stoch_k > 50
stochrsi_bull = cross_up or cross_up[1] or cross_up[2]
stochrsi_bear = cross_dn or cross_dn[1] or cross_dn[2]

// ══════════════════════════════════════════
//   E1 — BOLLINGER 0DTE · vela 9:45 · 3/4
//   Requisito: is_vela945
// ══════════════════════════════════════════
e1c_s = (inside_bb?1:0) + (above_vwap?1:0) + ((ema7>ema26)?1:0) + (v930_bull?1:0)
e1p_s = (inside_bb?1:0) + (below_vwap?1:0) + ((ema7<ema26)?1:0) + (v930_bear?1:0)
e1_call = is_vela945 and e1c_s >= min_e1
e1_put  = is_vela945 and e1p_s >= min_e1

// ══════════════════════════════════════════
//   E2 — RESUMEN 20 + H1 · 3/4
//   Requisitos: w_e2 + tendencia H1 definida
// ══════════════════════════════════════════
e2c_s = (inside_bb?1:0) + (above_vwap?1:0) + (no_res?1:0) + ((bull_solid and vol_high)?1:0)
e2p_s = (inside_bb?1:0) + (below_vwap?1:0) + (no_sup?1:0) + ((bear_solid and vol_high)?1:0)
e2_call = w_e2 and h1_bull and e2c_s >= min_e2
e2_put  = w_e2 and h1_bear and e2p_s >= min_e2

// ══════════════════════════════════════════
//   E3 — CIERRE CON DIRECCIÓN · 3/4
//   Requisitos: w_e3 + día movió >1%
//   SALIDA: 3:55 PM o SL 10%
// ══════════════════════════════════════════
e3c_s = (above_vwap?1:0) + (inside_bb?1:0) + ((ema7>ema26)?1:0) + ((close>open and vol_high)?1:0)
e3p_s = (below_vwap?1:0) + (inside_bb?1:0) + ((ema7<ema26)?1:0) + ((close<open and vol_high)?1:0)
e3_call = w_e3 and day_up and e3c_s >= min_e3
e3_put  = w_e3 and day_down and e3p_s >= min_e3

// ══════════════════════════════════════════
//   E4 — CONTINUACIÓN · 3/4
//   Requisito: w_e4
// ══════════════════════════════════════════
e4c_s = (above_vwap?1:0) + ((ema7>ema26)?1:0) + ((adx_v>adx_e4_min)?1:0) + ((mom_up and bull_solid and vol_high)?1:0)
e4p_s = (below_vwap?1:0) + ((ema7<ema26)?1:0) + ((adx_v>adx_e4_min)?1:0) + ((mom_down and bear_solid and vol_high)?1:0)
e4_call = w_e4 and e4c_s >= min_e4
e4_put  = w_e4 and e4p_s >= min_e4

// ══════════════════════════════════════════
//   E5 — SEMANAL · 4/5
//   Requisitos: w_e5 + tendencia diaria + rango semana previa <3%
// ══════════════════════════════════════════
e5c_s = (above_vwap?1:0) + ((ema7>ema26)?1:0) + (inside_bb?1:0) + ((adx_v>adx_e5_min)?1:0) + (stochrsi_bull?1:0)
e5p_s = (below_vwap?1:0) + ((ema7<ema26)?1:0) + (inside_bb?1:0) + ((adx_v>adx_e5_min)?1:0) + (stochrsi_bear?1:0)
e5_call = w_e5 and wk_bull and pw_range_ok and e5c_s >= min_e5
e5_put  = w_e5 and wk_bear and pw_range_ok and e5p_s >= min_e5
